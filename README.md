    function handleSearchSubmit(e, inputId) {
      e.preventDefault();
      const input = document.getElementById(inputId);
      if (!input) return;
      const q = input.value.trim().toLowerCase();
      const statusBar = document.getElementById('filter-status-bar');
      const statusText = document.getElementById('filter-status-text');

      if (!q) {
        statusBar.classList.add('hidden');
        renderProducts(PRODUCTS);
        return;
      }

      const filtered = PRODUCTS.filter(p =>
        p.name.toLowerCase().includes(q) ||
        p.brand.toLowerCase().includes(q) ||
        p.carFit.toLowerCase().includes(q) ||
        p.categoryLabel.toLowerCase().includes(q)
      );

      statusBar.classList.remove('hidden');
      statusText.textContent = `Kết quả tìm kiếm cho: "${input.value.trim()}"`;
      renderProducts(filtered);
      document.getElementById('san-pham-khuyen-mai').scrollIntoView({ behavior: 'smooth' });
    }

    // Điền nhanh thông tin sản phẩm vào Form Báo Giá
    function prefillQuote(productName, brand) {
      const sizeInput = document.getElementById('quote-size');
      const brandSelect = document.getElementById('quote-brand');
      if (sizeInput) sizeInput.value = productName;
      if (brandSelect) {
        for (let i = 0; i < brandSelect.options.length; i++) {
          if (brandSelect.options[i].value.toLowerCase().includes(brand.toLowerCase())) {
            brandSelect.selectedIndex = i;
            break;
          }
        }
      }
      document.getElementById('form-bao-gia').scrollIntoView({ behavior: 'smooth' });
      setTimeout(() => {
        const nameInput = document.getElementById('quote-name');
        if (nameInput) nameInput.focus();
      }, 500);
    }

    let currentSlide = 0;
    const totalSlides = 2;
    let slideInterval = null;

    function showSlide(index) {
      currentSlide = (index + totalSlides) % totalSlides;
      const slides = document.querySelectorAll('.hero-slide');
      slides.forEach((sl, idx) => {
        if (idx === currentSlide) {
          sl.classList.remove('opacity-0', 'z-0', 'pointer-events-none');
          sl.classList.add('opacity-100', 'z-10');
        } else {
          sl.classList.remove('opacity-100', 'z-10');
          sl.classList.add('opacity-0', 'z-0', 'pointer-events-none');
        }
      });

      const dot0 = document.getElementById('slider-dot-0');
      const dot1 = document.getElementById('slider-dot-1');
      if (dot0 && dot1) {
        if (currentSlide === 0) {
          dot0.className = 'w-6 h-2 rounded-full bg-brand-red transition-all';
          dot1.className = 'w-2.5 h-2 rounded-full bg-white/60 hover:bg-white transition-all';
        } else {
          dot1.className = 'w-6 h-2 rounded-full bg-brand-red transition-all';
          dot0.className = 'w-2.5 h-2 rounded-full bg-white/60 hover:bg-white transition-all';
        }
      }
    }

    function changeSlide(step) {
      showSlide(currentSlide + step);
      resetSlideTimer();
    }

    function goToSlide(index) {
      showSlide(index);
      resetSlideTimer();
    }

    function resetSlideTimer() {
      if (slideInterval) clearInterval(slideInterval);
      slideInterval = setInterval(() => {
        showSlide(currentSlide + 1);
      }, 4500);
    }

    // Khởi tạo giao diện khi tải trang
    window.addEventListener('DOMContentLoaded', () => {
      renderProducts(PRODUCTS);
      renderBlogPosts('all');
      resetSlideTimer();

      // Hỗ trợ vuốt cảm ứng trên Banner điện thoại
      const sliderWrapper = document.getElementById('hero-slider-wrapper');
      let touchStartX = 0;
      if (sliderWrapper) {
        sliderWrapper.addEventListener('touchstart', e => {
          touchStartX = e.changedTouches[0].screenX;
        }, { passive: true });
        sliderWrapper.addEventListener('touchend', e => {
          const diff = e.changedTouches[0].screenX - touchStartX;
          if (Math.abs(diff) > 40) {
            changeSlide(diff < 0 ? 1 : -1);
          }
        }, { passive: true });
      }

      // Điều khiển Menu Drawer & Tìm kiếm trên điện thoại
      const mobileMenuBtn = document.getElementById('mobile-menu-btn');
      const mobileDrawer = document.getElementById('mobile-drawer');
      const mobileOverlay = document.getElementById('mobile-drawer-overlay');
      const mobileClose = document.getElementById('mobile-drawer-close');

      function toggleMobileMenu(show) {
        if (show) {
          mobileOverlay.classList.remove('hidden');
          mobileDrawer.classList.remove('-translate-x-full');
        } else {
          mobileOverlay.classList.add('hidden');
          mobileDrawer.classList.add('-translate-x-full');
        }
      }

      if (mobileMenuBtn) mobileMenuBtn.addEventListener('click', () => toggleMobileMenu(true));
      if (mobileClose) mobileClose.addEventListener('click', () => toggleMobileMenu(false));
      if (mobileOverlay) mobileOverlay.addEventListener('click', () => toggleMobileMenu(false));
      document.querySelectorAll('.mobile-nav-link').forEach(link => {
        link.addEventListener('click', () => toggleMobileMenu(false));
      });

      const mobileSearchToggle = document.getElementById('mobile-search-toggle');
      const mobileSearchBar = document.getElementById('mobile-search-bar');
      if (mobileSearchToggle && mobileSearchBar) {
        mobileSearchToggle.addEventListener('click', () => {
          mobileSearchBar.classList.toggle('hidden');
        });
      }
    });
  </script>
